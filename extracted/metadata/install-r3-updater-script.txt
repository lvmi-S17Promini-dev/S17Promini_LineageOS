ui_print("LineageOS 19.1 - S17Promini full install");
assert(getprop("ro.product.device") == "S17Promini" || getprop("ro.product.vendor.device") == "S17Promini");
ifelse(is_mounted("/system_root"), unmount("/system_root"), "");
ifelse(is_mounted("/system"), unmount("/system"), "");
ifelse(is_mounted("/vendor"), unmount("/vendor"), "");
assert(update_dynamic_partitions(package_extract_file("dynamic_partitions_op_list")));
ui_print("Writing system");
assert(block_image_update(map_partition("system"), package_extract_file("system.transfer.list"), "system.new.dat", "system.patch.dat"));

ui_print("Writing vendor");
assert(block_image_update(map_partition("vendor"), package_extract_file("vendor.transfer.list"), "vendor.new.dat", "vendor.patch.dat"));
assert(range_sha1(map_partition("vendor"), "2,0,348632") == "4ef961e7535844dcf30ad0404d9b3a9cb4f96db8");
unmap_partition("vendor");
assert(package_extract_file("recovery.img", "/dev/block/by-name/recovery"));
